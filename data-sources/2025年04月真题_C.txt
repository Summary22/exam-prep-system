struct stu
{
    int no;
    const char *name;
    int score;
};
struct stu *py(struct stu *p)
{
    struct stu *s = p;
    for (p++;p->no > 0; p++)
        if (s->score < p->score)
            s=p;
    return s;
}
void main()
{
    struct stu student[]={ {1001, "Sun", 76}, {1002, "Ling", 88},

    printf("NO:%d NAME:%s SCORE: %d\n", p->no, p->name, p->score);
}
